%% One data spine, two model arms that differ only in whether they consume
%% training data, one scoring spine, and a regime axis only the encoders traverse.

\begin{figure*}[t]
\centering
\begin{tikzpicture}[
  font=\footnotesize,
  >={Stealth[length=3.6pt,width=2.8pt]},
  bx/.style   ={draw=rule, line width=0.4pt, rounded corners=1.2pt, fill=white,
                align=center, text width=23mm, inner xsep=2pt, inner ysep=3pt,
                font=\scriptsize},
  src/.style  ={bx, fill=panel},
  enc/.style  ={bx, draw=encblue, line width=0.9pt},
  llm/.style  ={bx, draw=llmgold, line width=0.9pt, dashed, dash pattern=on 2pt off 1.4pt},
  ed/.style   ={->, draw=rule, line width=0.5pt},
  encd/.style ={->, draw=encblue, line width=0.9pt},
  llmd/.style ={->, draw=llmgold, line width=0.9pt, dashed,
                dash pattern=on 2pt off 1.4pt},
  tag/.style  ={font=\scriptsize, text=black!60, inner sep=1.2pt},
]
\def\cA{0}\def\cB{2.92}\def\cC{5.84}\def\cD{9.05}\def\cE{12.15}\def\cF{15.25}

%% --- data spine ----------------------------------------------------------
\node[src] (s1)  at (\cA, 0.62) {\textbf{\pex{}}~\cite{lima2019}\\[0.5pt]969\,$\to$\,968\\47 projects};
\node[src] (s2)  at (\cA,-0.62) {\textbf{SecReq}~\cite{knauss2011}\\[0.5pt]510\,$\to$\,444\\3 specifications};
\node[src] (uni) at (\cB, 0)    {\textbf{Unified corpus}\\[0.5pt]1{,}412 requirements\\50 project groups\\67 duplicates dropped};
\node[src] (spl) at (\cC, 0)    {\textbf{Frozen splits}\\[0.5pt]14 families\\107 folds, seed 42\\SHA-256-pinned ids};

\draw[ed] (s1.east) -- ++(0.20,0) |- (uni.west);
\draw[ed] (s2.east) -- ++(0.20,0) |- (uni.west);
\draw[ed] (uni.east) -- (spl.west);

%% --- the two model arms --------------------------------------------------
\node[enc] (arm1) at (\cD, 1.04) {\textbf{Encoder arm}\\[0.5pt]BERT, RoBERTa\\$\pm$ class weighting\\\emph{re-fitted per regime}};
\node[llm] (arm2) at (\cD,-1.04) {\textbf{Prompted arm}\\[0.5pt]8 LLMs, 3 tiers\\zero- and few-shot\\\emph{never fitted}};

\draw[encd] (spl.east) -- ++(0.42,0) |- (arm1.west);
\draw[llmd] (spl.east) -- ++(0.42,0) |- (arm2.west);

%% --- shared scoring spine ------------------------------------------------
\node[src] (sco) at (\cE, 0) {\textbf{Strict scoring}\\[0.5pt]identical test items\\one-word parser;\\unparseable $=$ wrong\\76{,}115 predictions\\gate: 81 of 87 cells};
\node[src] (ana) at (\cF, 0) {\textbf{Analysis}\\[0.5pt]\Fm{} $+$ bootstrap CI\\260 paired McNemar\\Holm and BH, $\alpha{=}0.05$\\cost, $N^{\star}{=}C_0/\Delta c$};

\draw[encd] (arm1.east) -| ($(sco.north)+(0,0.34)$) -- (sco.north);
\draw[llmd] (arm2.east) -| ($(sco.south)-(0,0.34)$) -- (sco.south);
\draw[ed]   (sco.east) -- (ana.west);

%% --- the regime axis -----------------------------------------------------
\begin{scope}[shift={(0,-2.50)}]
  \fill[panel, rounded corners=1.2pt] (-1.32,-0.40) rectangle (16.52,0.42);

  \foreach \x/\name/\desc in {%

\node[tag, anchor=north west, align=left, text width=176mm] at (-1.32,-3.02)
  {\textbf{Solid (upper) path:} consumes training folds, re-fitted at every
   station of the regime axis. \textbf{Dashed (lower) path:} consumes only test
   items, so it is regime-invariant and forms the fixed reference against which
   the encoders' drop is measured.};

\end{tikzpicture}
\caption{Experimental architecture. One data spine (left) feeds two model arms
that differ in exactly one respect: whether they consume training data. Both
answer the \emph{same} items---so every encoder--LLM comparison is paired
item-for-item---and both pass through one strict scoring spine (right). The band
beneath is the study's independent variable: the distribution shift between what
a model was fitted on and what it is scored on.}
\label{fig:arch}
\end{figure*}
